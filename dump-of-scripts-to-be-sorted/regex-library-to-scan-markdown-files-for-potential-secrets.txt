############################################################
# CATEGORY 1: AUTHENTICATION / AUTHORIZATION
############################################################

# Bearer tokens (require space + high-entropy token)
Bearer\s+[A-Za-z0-9._~+/=]{20,}

# JWT tokens (exclude paths and method chains)
(?<!\/)(?![^\/]*\/)[A-Za-z0-9_-]{10,}\.[A-Za-z0-9_-]{10,}\.[A-Za-z0-9_-]{10,}

# Authorization header with Bearer token
Authorization\s*:\s*Bearer\s+[A-Za-z0-9._~+/=]{20,}

# OAuth access tokens (anchored to key name)
access_token\s*[:=]\s*["']?[A-Za-z0-9._~+/=]{20,}

# OAuth refresh tokens
refresh_token\s*[:=]\s*["']?[A-Za-z0-9._~+/=]{20,}

############################################################
# CATEGORY 2: SECRETS (API KEYS, CLIENT SECRETS, TOKENS)
############################################################

# Generic secrets (mixed-case + digits, exclude paths/namespaces) - check output without this.
# (?<!\/)(?![^\/]*\/)(?=.*\d)(?=.*[A-Z])(?:[A-Za-z0-9]{12,}[A-Za-z0-9_\-+=]*){2,}


# API keys with common prefixes (anchored)
(api[_-]?key|apikey|apiKey)\s*[:=]\s*["']?[A-Za-z0-9._~+/=]{20,}

# Client secrets (anchored)
(client[_-]?secret|clientSecret)\s*[:=]\s*["']?[A-Za-z0-9._~+/=]{20,}

# Encrypted or hashed secrets (base64-like, avoid paths) - check output without this.
(?<!\/)(?![^\/]*\/)[A-Za-z0-9+/]{24,}={0,2}

############################################################
# CATEGORY 3: IDENTIFIERS / PII
############################################################

# Email addresses
# [A-Za-z0-9._%+-]+@[A-Za-z0-9.-]+\.[A-Za-z]{2,}

# Indian mobile numbers (10 digits, starting with 6–9)
# \b[6-9]\d{9}\b

# International phone numbers (very loose)
# \+\d{1,3}[\s-]?\d{7,14}

# Usernames or IDs (anchored to key name)
(username|user_id|userid|userId)\s*[:=]\s*["']?[A-Za-z0-9._-]{4,}


############################################################
# CATEGORY 4: INFRASTRUCTURE IDENTIFIERS
############################################################

# IPv4 addresses
# \b(?:\d{1,3}\.){3}\d{1,3}\b

# URLs with embedded credentials
https?:\/\/[^\/\s]+:[^\/\s]+@[^\/\s]+

# Hostnames (avoid paths)
# \b[a-zA-Z0-9.-]+\.(com|net|org|io|internal|corp)\b
